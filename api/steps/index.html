    Step operators¶

    Step operators are the building blocks of a first-order method, covering affine steps, ReLU, saturated linear, and soft thresholding.

    Step ¶

    A base class for thresholding and activation steps. This class defines a common interface for step-like operators that transform inputs element-wise.

    A step represents the relation: output = f(arg), where f is the step function (e.g., ReLU, soft threshold, saturated linear).

    Attributes:

    Name Type Description
    arg LinearExpression

    Right-hand side (input) linear expression.
    output LinearExpression

    Left-hand side (output) linear expression. Bounds on the output are stored in output.lb and output.ub.
    relax_binary_vars bool

    If True, binary decision variables associated with the operator are relaxed to take continuous values.
    skip_obbt bool

    If True, OBBT bound tightening is skipped for this step type. Override in subclasses that compute bounds analytically.
    obbt_output bool

    If true, OBBT is also run on the step's output leaves once the step's relaxation is in the OBBT LP. Enabled for Affine with non-identity output_mat, whose output bounds cannot otherwise see the tightened input box without M^{-1}.
    Notes

    Input vs output bounds: - step.lb, step.ub: bounds on the INPUT (arg) - step.output.lb, step.output.ub: bounds on the OUTPUT (output)

    When update_bounds(lb, ub) is called with tightened input bounds, subclasses should update both self.lb/self.ub (input) and self.output.lb/self.output.ub (output, by applying the step function).

    class Step:
    """
    A base class for thresholding and activation steps. This class
    defines a common interface for step-like operators that transform
    inputs element-wise.

    A step represents the relation: output = f(arg), where f is the step
    function (e.g., ReLU, soft threshold, saturated linear).

    Attributes
    ----------
    arg : LinearExpression
        Right-hand side (input) linear expression.
    output : LinearExpression
        Left-hand side (output) linear expression. Bounds on the output
        are stored in ``output.lb`` and ``output.ub``.
    relax_binary_vars : bool
        If True, binary decision variables associated with the operator
        are relaxed to take continuous values.
    skip_obbt : bool
        If True, OBBT bound tightening is skipped for this step type.
        Override in subclasses that compute bounds analytically.
    obbt_output : bool
        If true, OBBT is also run on the step's output leaves once the step's
        relaxation is in the OBBT LP. Enabled for `Affine` with non-identity
        ``output_mat``, whose output bounds cannot otherwise see the tightened
        input box without ``M^{-1}``.

    Notes
    -----
    Input vs output bounds:
        - ``step.lb``, ``step.ub``: bounds on the INPUT (arg)
        - ``step.output.lb``, ``step.output.ub``: bounds on the OUTPUT (output)

    When ``update_bounds(lb, ub)`` is called with tightened input bounds,
    subclasses should update both ``self.lb``/``self.ub`` (input) and
    ``self.output.lb``/``self.output.ub`` (output, by applying the step function).
    """

    skip_obbt: bool = False
    obbt_output: bool = False

    def __init__(self, arg: LinearExpression, relax_binary_vars: bool = False):
        self.arg = arg
        self.relax_binary_vars = relax_binary_vars
        # Default: every output coordinate is "projected" (i.e., the step
        # transformation applies to all rows). Subclasses that support
        # partial projection (e.g. ``Relu`` with ``proj_ranges``) override
        # this in their own ``__init__``.
        self.proj_indices = set(range(arg.ndim))

    @cached_property
    def output(self) -> LinearExpression:
        """Left-hand side linear expression in the operator definition."""
        return LinearExpression.from_int(self.arg.ndim)

    @property
    def output_leaves(self) -> list[LinearExpression]:
        """Leaves this step introduces and that must be registered with the
        canonicalizer/step decomposer.

        Default: ``[self.output]`` (the output is itself a leaf). Steps that expose
        a non-leaf composite ``output`` (currently only partial-projection
        ``Relu``) override this to return the smaller fresh leaf actually
        introduced.
        """
        return [self.output]

    def update_bounds(self, lb, ub):
        raise NotImplementedError

    # TODO: figure out a better way to hash these objects
    def __hash__(self):
        return hash((self.arg, self.output))

    class Affine(Step):
    r"""
    An affine operator of the following form:

    $$
      \phi(v,x) = \{ w \in \mathbf{R}^d \mid Mw = By,\ \ y = (v,x) \},
    $$

    where $M \in \mathbf{R}^{ d \times d }$ is an invertible matrix and
    $B \in \mathbf{R}^{d \times (p+d)}$.

    It can also be constructed via ``weight`` and optional ``bias`` to represent
    the explicit transform $w = W \cdot \text{arg} + b$, where ``arg`` is the
    input expression. In this form ``output_mat`` is set to identity internally.

    Attributes
    ----------
    output: LinearExpression
        Left-hand side linear expression in the operator definition.
    arg: LinearExpression
        Right-hand side linear expression in the operator definition.
    lb: np.ndarray
        Lower bounds on the INPUT (arg), before the affine transform.
    ub: np.ndarray
        Upper bounds on the INPUT (arg), before the affine transform.
    relax_binary_vars: bool
        If True, binary decision variables associated with the operator
        are relaxed to take continuous values.
    output_mat: np.matrix
        Coefficient matrix on the left-hand side of the operator equation.
    obbt_output: bool
        True when ``output_mat`` is not the identity: OBBT is then also run on the
        output leaf after the affine equality is added to the OBBT LP (see
        ``StepDecomposer._process_primitive``). For identity ``output_mat`` the
        output equals the arg, so ``sync_output_bounds`` already copies the
        OBBT-tightened input box and the extra LPs would be redundant.
    """

    def __init__(
        self,
        arg: LinearExpression,
        output_mat: np.matrix = None,
        weight: np.ndarray = None,
        bias: np.ndarray = None,
        relax_binary_vars: bool = False,
    ):
        if weight is not None:
            arg = weight @ arg
            if bias is not None:
                arg = arg + bias
            output_mat = np.eye(weight.shape[0])
        elif output_mat is None:
            raise ValueError("Either output_mat or weight must be provided.")
        super().__init__(arg, relax_binary_vars)
        self.output_mat = output_mat
        self.lb, self.ub = self.arg.bound_propagation()
        # from_int defaults lb/ub to scalars; force array form so subsequent
        # steps' bound_propagation() does not hit sparse @ scalar NaN errors.
        ndim = self.arg.ndim
        self.output.update_bounds(np.full(ndim, -np.inf), np.full(ndim, np.inf))

    @cached_property
    def obbt_output(self) -> bool:
        return not is_identity_matrix(self.output_mat)

    def update_bounds(self, lb: float | np.ndarray, ub: float | np.ndarray):
        """
        Update INPUT bounds (on arg) and propagate analytically to OUTPUT bounds on output.
        """
        self.lb, self.ub = lb, ub
        self.sync_output_bounds()

    def sync_output_bounds(self) -> None:
        """
        Tighten output bounds from the current input bounds.

        Interval-propagates ``output_mat_inverse`` (built by linear solves, never an
        explicit inverse). When ``output_mat`` is the identity, ``output == arg``, so
        the (possibly OBBT-tightened) input box ``self.lb``/``self.ub`` also bounds the
        output directly. ``output.update_bounds`` intersects with the existing output
        bounds, so tightening applied earlier is never discarded. For general
        ``output_mat`` the tightened input box is exploited by OBBT on the output leaf
        (see ``StepDecomposer._process_primitive``).
        """
        out_lb, out_ub = self.output_mat_inverse.bound_propagation()
        if is_identity_matrix(self.output_mat):
            out_lb = np.maximum(out_lb, self.lb)
            out_ub = np.minimum(out_ub, self.ub)
        self.output.update_bounds(lb=out_lb, ub=out_ub)
        # Repair the *combined* result (update_bounds intersects with the existing
        # output bounds, which is where noise crossings arise). Repairing before
        # update_bounds would be undone by the intersection, hence the direct write;
        # update_bounds is a plain elementwise max/min with no other side effects.
        self.output.lb, self.output.ub = repair_crossed_bounds(self.output.lb, self.output.ub)

    @cached_property
    def output_mat_inverse(self) -> LinearExpression:
        r"""
        Return a `LinearExpression` formed by applying $A^{-1}$, where $A$ is
        ``self.output_mat``, to each right-hand-side component $b_k$—i.e.,
        solving $A x_k = b_k$ for all $k$.
        Supports both dense and sparse ``output_mat``.
        """

        def _to_dense(b):
            return np.asarray(b.todense()) if spa.issparse(b) else np.asarray(b)

        if spa.issparse(self.output_mat):
            lu = spa.linalg.splu(self.output_mat.tocsc())

            def solve(b):
                return lu.solve(_to_dense(b))
        else:

            def solve(b):
                return np.asarray(np.linalg.solve(self.output_mat, _to_dense(b)))

        return LinearExpression(
            decomposition_dict={
                k: np.squeeze(solve(v)) for k, v in self.arg.decomposition_dict.items()
            },
            const=np.squeeze(solve(self.arg.const)),
        )

    class Relu(Step):
    r"""
    Rectified Linear Unit (ReLU) step operator, defined as:

    $$
        (\phi(v,x))_i = \mathbf{ReLU}(a^T y) = \max\{a^T y, 0\}.
    $$

    It sets all negative values to zero while keeping the positive ones unchanged.

    Attributes
    ----------
    arg : LinearExpression
        Right-hand side (input) linear expression.
    lb : np.ndarray
        Lower bounds on the INPUT (arg), before ReLU is applied.
    ub : np.ndarray
        Upper bounds on the INPUT (arg), before ReLU is applied.
    relax_binary_vars : bool
        If True, binary decision variables associated with the operator
        are relaxed to take continuous values.
    idx_with_binary_vars : set[int]
        Indices corresponding to binary decision variables.
    proj_indices : set[int]
        Indices of output coordinates subjected to ReLU projection.

    Notes
    -----
    Partial projection (``proj_ranges`` covers only a subset of the input
    coordinates) is encoded with a *composite* ``output``: only
    ``len(proj_indices)`` fresh leaf variables are introduced (stored on
    ``self._output_proj_leaf``), and the non-projected output coordinates
    share decomposition with ``arg`` directly. This keeps the SDP variable
    count, MIP variable count, and simulator leaf count at the minimum.
    For the full-projection default, ``output`` remains a single leaf of size
    ``arg.ndim`` (current behavior).
    """

    def __init__(
        self,
        arg: LinearExpression,
        proj_ranges=None,
        relax_binary_vars=False,
    ):
        super().__init__(arg, relax_binary_vars)
        self.proj_indices = (
            set(range(0, self.arg.ndim)) if proj_ranges is None else self._get_ranges(proj_ranges)
        )
        self.idx_with_binary_vars = set()
        # When True, the SDP canonicalizer does not register this ReLU as an
        # orthant prox for the subgradient-monotonicity cut. Set by decomposers
        # that build a composite operator (e.g. soft-threshold) from internal
        # ReLUs and register the composite prox instead. Declared here as a formal
        # attribute (rather than a dynamic one) so it survives copying.
        self._skip_prox_registration = False

        arg_lb, arg_ub = self.arg.bound_propagation()

        if self.proj_indices != set(range(self.arg.ndim)):
            # Initialize the proj-leaf with array-valued bounds so the composite
            # output's bound_propagation has a consistent shape to work with.
            proj_list = sorted(self.proj_indices)
            proj_lb = np.maximum(0, np.asarray(arg_lb)[proj_list])
            proj_ub = np.maximum(0, np.asarray(arg_ub)[proj_list])
            self._output_proj_leaf = LinearExpression.from_int(
                len(self.proj_indices), lb=proj_lb, ub=proj_ub
            )
            # Override Step.output cached_property with a composite expression
            # whose non-proj rows share storage with arg and whose proj rows
            # come from _output_proj_leaf.
            self.__dict__["output"] = self._build_composite_output()
        else:
            self._output_proj_leaf = None

        self.update_bounds(arg_lb, arg_ub)

    def __hash__(self):
        return hash((self.arg, self.output))

    @staticmethod
    def _get_ranges(ranges: Tuple[int, int] | List[Tuple[int, int]]) -> set[int]:
        """
        Return the set of integers covered by one or more half-open intervals.

        Parameters
        ----------
        ranges: Tuple[int, int] or List[Tuple[int, int]]
            Either a single (start, end) pair or an iterable of such pairs.
            Each pair is interpreted as a half-open interval ``[start, end)``.

        Returns
        -------
        set[int]
            The unique integers contained in the given half-open range(s).

        Examples
        --------
        >>> Relu._get_ranges(((1, 2), (5, 9)))
        {1, 5, 6, 7, 8}

        >>> Relu._get_ranges((4, 7))
        {4, 5, 6}
        """
        # Try the iterable-of-pairs form first. If ``ranges`` is a single
        # ``(start, end)`` pair, unpacking its int elements raises TypeError
        # ("cannot unpack non-iterable int") — fall back to the single-range
        # form. This avoids the pre-existing ``ranges.append((0, 0))``
        # workaround which silently mutated caller-owned lists.
        try:
            return {i for (start, end) in ranges for i in range(start, end)}
        except TypeError:
            return set(range(ranges[0], ranges[1]))

    def _build_composite_output(self) -> LinearExpression:
        """Construct the composite output: proj rows from ``_output_proj_leaf``,
        non-proj rows passed through directly from ``arg``.

        The composite has dimension ``arg.ndim`` and its decomposition_dict
        references both ``_output_proj_leaf`` and the leaves of ``arg``.
        """
        n = self.arg.ndim
        proj_list = sorted(self.proj_indices)
        nonproj_list = sorted(set(range(n)) - self.proj_indices)
        k = len(proj_list)

        # Selection matrix placing _output_proj_leaf (size k) into proj rows of output.
        S_proj = spa.csc_matrix(
            (np.ones(k), (np.array(proj_list, dtype=int), np.arange(k))),
            shape=(n, k),
        )

        # Selection matrix masking out proj rows of arg; non-proj rows of arg
        # pass through to output unchanged.
        m = len(nonproj_list)
        if m > 0:
            nonproj_arr = np.array(nonproj_list, dtype=int)
            S_nonproj = spa.csc_matrix(
                (np.ones(m), (nonproj_arr, nonproj_arr)),
                shape=(n, n),
            )
        else:
            S_nonproj = spa.csc_matrix((n, n))

        decomposition_dict: dict = {self._output_proj_leaf: S_proj}
        for leaf, coeff in self.arg.decomposition_dict.items():
            composed = S_nonproj @ coeff
            if leaf in decomposition_dict:
                decomposition_dict[leaf] = decomposition_dict[leaf] + composed
            else:
                decomposition_dict[leaf] = composed

        const = S_nonproj @ self.arg.const

        # Drop entries whose composed coefficient is all-zero — happens when
        # an arg leaf contributes only to projected rows, since ``S_nonproj``
        # zeros out those rows. Pruning prevents downstream canonicalizers
        # from looking up an irrelevant leaf in ``var_idx_map``.
        result = LinearExpression(decomposition_dict=prune_dict(decomposition_dict), const=const)
        # Initialize array-valued bounds so later update_bounds calls have a
        # consistent shape to maximize/minimize against.
        result.lb, result.ub = result.bound_propagation()
        return result

    def relu(self, x: np.ndarray) -> np.ndarray:
        """
        Apply ReLU activation selectively to specified indices.

        Parameters
        ----------
        x: np.ndarray
            A 1-D array of values on which to apply the activation.

        Returns
        -------
        np.ndarray
            A new array where elements at positions specified in
            ``self.proj_indices`` are passed through ReLU, while
            all other elements are left unchanged.
        """
        proj_mask = np.isin(np.arange(len(x)), list(self.proj_indices))
        return np.where(proj_mask, np.maximum(0, x), x)

        the output is itself a single fresh leaf.

        return [self.output]

    class QuadraticProgram(Step):
    r"""
    Step representing the solution of an arbitrary convex quadratic program

    $$
        x^\star = \arg\min_x\; \tfrac12 x^\top P x + q^\top x
        \quad\text{s.t.}\quad A x = b,\; C x \le d,
    $$

    where $P \in \mathbf{R}^{n\times n}$ is a fixed **symmetric PSD** matrix
    (dense ``numpy`` or ``scipy.sparse``), and ``q``, ``b``, ``d`` may be numeric
    data or `LinearExpression` parameters (carrying dependence on prior
    iterates). The verification-side lowering (``decompose_quadraticprogram`` in
    the step decomposer) posts the KKT system

    $$
        P x + q + A^\top \nu + C^\top \lambda = 0,\quad
        A x = b,\quad C x \le d,\quad
        0 \le \lambda \perp (d - C x) \ge 0,
    $$

    with $\nu$ a free equality multiplier and $\lambda \ge 0$, using
    the complementarity primitive. The MIP big-M encoding of the complementarity
    needs a finite upper bound on $\lambda$; supply it via ``dual_ub``
    whenever inequalities are present (OBBT tightens it further when enabled).

    Invoked via `Verifier.quadratic_program` rather than the single-``arg``
    ``run_step`` path: an arbitrary QP has up to three parametric inputs
    (``q``, ``b``, ``d``), and each ``LinearExpression`` already references the
    prior iterates it was built from, so no input is forced into an ``arg`` slot.

    Parameters
    ----------
    P : np.ndarray or scipy.sparse matrix
        The ``(n, n)`` quadratic term; must be symmetric and PSD.
    q : None, float, np.ndarray, or LinearExpression, optional
        Linear cost of dimension ``n``. If ``None`` (projection convenience),
        ``q`` defaults to ``-arg`` in the decomposer, which requires ``arg`` to be
        a dim-``n`` ``LinearExpression``.
    A : np.ndarray or scipy.sparse, optional
        Equality matrix of shape ``(p, n)``. ``None`` for no equality constraints.
    b : float, np.ndarray, or LinearExpression, optional
        Equality right-hand side of dimension ``p``.
    C : np.ndarray or scipy.sparse, optional
        Inequality matrix of shape ``(m, n)``. ``None`` for no inequalities.
    d : float, np.ndarray, or LinearExpression, optional
        Inequality right-hand side of dimension ``m``.
    dual_ub : None, float, or np.ndarray, optional
        Finite upper bound(s) on the inequality dual ``lambda >= 0``. Required
        (raised eagerly in the decomposer) whenever ``C`` is present.
    primal_bounds : tuple[np.ndarray, np.ndarray], optional
        ``(lb, ub)`` on the primal ``x``. Needed when ``P`` is singular / not
        positive definite (the stationarity row cannot pin ``x`` bounds);
        otherwise derived analytically and tightened by OBBT.
    nu_bounds : tuple[np.ndarray, np.ndarray], optional
        ``(lb, ub)`` on the free equality multiplier ``nu``. Defaults to
        ``+/- inf`` (``nu`` only appears in the linear stationarity row).
    arg : LinearExpression, optional
        The projection point used when ``q is None`` (then ``q := -arg``).
        Optional otherwise; retained only as the nominal Step input.
    relax_binary_vars : bool, default False
        Forwarded to the Step interface (relax big-M binaries when set).
    """

    skip_obbt: bool = True

    def __init__(
        self,
        P,
        q=None,
        A=None,
        b=None,
        C=None,
        d=None,
        dual_ub=None,
        primal_bounds=None,
        nu_bounds=None,
        arg: LinearExpression = None,
        relax_binary_vars: bool = False,
    ):
        # --- validate P: square, symmetric, PSD (dense or sparse) ---
        if P.ndim != 2 or P.shape[0] != P.shape[1]:
            raise ValueError(f"P must be a square 2-D matrix; got shape {P.shape}")
        if spa.issparse(P):
            P = P.tocsr()  # CSR supports T, arithmetic, and @ (LinearExpression.__rmatmul__)
        n = P.shape[0]
        sym_err = _sym_error(P)
        if sym_err > _PSD_TOL:
            raise ValueError(f"P must be symmetric; max|P - P^T| = {sym_err:.3e}")
        min_eig = _min_eig(P)
        if min_eig < -_PSD_TOL:
            raise ValueError(
                f"P must be positive semidefinite; smallest eigenvalue {min_eig:.3e} < -{_PSD_TOL:g}"
            )

        # --- validate parametric data dimensions ---
        if q is None:
            if arg is None or not _is_linexpr(arg) or arg.ndim != n:
                raise ValueError(
                    "q=None defaults to -arg, which requires arg to be a "
                    f"LinearExpression of dimension n={n}."
                )
        elif _vec_dim(q, n) != n:
            raise ValueError(f"q dim {_vec_dim(q, n)} must match P dimension n={n}")

        if A is not None:
            if A.shape[1] != n:
                raise ValueError(f"A must have {n} columns to match P; got shape {A.shape}")
            p = A.shape[0]
            if b is None:
                raise ValueError("b must be provided when A is given")
            if _vec_dim(b, p) != p:
                raise ValueError(f"b dim {_vec_dim(b, p)} must match A rows {p}")
        if C is not None:
            if C.shape[1] != n:
                raise ValueError(f"C must have {n} columns to match P; got shape {C.shape}")
            m = C.shape[0]
            if d is None:
                raise ValueError("d must be provided when C is given")
            if _vec_dim(d, m) != m:
                raise ValueError(f"d dim {_vec_dim(d, m)} must match C rows {m}")

        # Bypass Step.__init__ (it dereferences arg.ndim): arg is optional here.
        self.arg = arg
        self.relax_binary_vars = relax_binary_vars
        self.n = n
        self.proj_indices = set(range(n))

        self.P = P
        self.q = q
        self.A = A
        self.b = b
        self.C = C
        self.d = d
        self.dual_ub = dual_ub
        self.primal_bounds = primal_bounds
        self.nu_bounds = nu_bounds

    @cached_property
    def output(self) -> LinearExpression:
        """Output is the QP primal solution ``x`` of dimension ``n`` (``!= arg.ndim``)."""
        return LinearExpression.from_int(self.n)

    def update_bounds(self, lb, ub):
        """Store primal bounds (interface compliance; QP is a composite step that
        bypasses ``_process_primitive``, so this is rarely invoked)."""
        self.lb, self.ub = lb, ub
        self.output.update_bounds(lb=lb, ub=ub)

    def __hash__(self):
        return hash((self.arg, self.output))

    def solve(self, q=None, b=None, d=None) -> np.ndarray:
        r"""
        Numerically solve the QP with cvxpy (reference forward-evaluation).

        Numeric overrides for ``q``/``b``/``d`` may be supplied for the case where
        the stored values are ``LinearExpression`` parameters; otherwise the stored
        numeric data is used. ``P`` is densified for cvxpy.

        Returns
        -------
        np.ndarray
            The optimal primal ``x`` of dimension ``n``.
        """
        import cvxpy as cp

        def _resolve(override, stored, name):
            value = override if override is not None else stored
            if value is None:
                return None
            if _is_linexpr(value):
                raise ValueError(
                    f"{name} is a LinearExpression; pass a numeric override to solve()"
                )
            return np.asarray(value, dtype=float)

        P_dense = P.toarray() if spa.issparse(P := self.P) else np.asarray(P, dtype=float)
        q_val = _resolve(q, self.q, "q")
        if q_val is None:
            raise ValueError("q is None (defaults to -arg); pass a numeric q to solve()")
        q_val = np.broadcast_to(q_val, (self.n,)).astype(float)

        x = cp.Variable(self.n)
        objective = q_val @ x
        if np.any(P_dense):
            objective = 0.5 * cp.quad_form(x, cp.psd_wrap(P_dense)) + objective
        constraints = []
        if self.A is not None:
            A = self.A.toarray() if spa.issparse(self.A) else np.asarray(self.A, dtype=float)
            b_val = _resolve(b, self.b, "b")
            constraints.append(A @ x == np.broadcast_to(b_val, (A.shape[0],)))
        if self.C is not None:
            C = self.C.toarray() if spa.issparse(self.C) else np.asarray(self.C, dtype=float)
            d_val = _resolve(d, self.d, "d")
            constraints.append(C @ x <= np.broadcast_to(d_val, (C.shape[0],)))
        # Pin the conic solver (cvxpy's default may be an unavailable/expired backend).
        cp.Problem(cp.Minimize(objective), constraints).solve(solver=cp.CLARABEL)
        return np.asarray(x.value, dtype=float)
